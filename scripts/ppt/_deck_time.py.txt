
# ⚠️ 챕터 표지(The problem)는 org_intro_deck.py 가 «한 번만» 부른다.
#    여기서 부르면 both 판에서 같은 표지가 두 번 나온다(2026-09-11 정정).

# ── 병기가 결과를 가른다 — 국가 통계로 「늦으면 의술로도 못 메운다」를 세운다
s = B.content(
    prs, "WHY EARLY",
    "병기가 결과를 가릅니다",
    "위암 5년 상대생존율입니다. 같은 진단인데 발견 시점에 따라 결과가 갈립니다.",
)
cw3 = (W - M * 2 - 20 * 2) / 3
for i, (v, l, sub, acc) in enumerate([
    ("97.6%", "국한 (요약병기)", "종양이 장기를 벗어나지 않은 상태", False),
    ("7.5%", "원격전이", "다른 장기로 퍼진 뒤 발견", False),
    ("13배", "병기가 만든 차이", "늦게 발견되면 의술로도 메우기 어렵습니다", True),
]):
    B.stat(s, M + i * (cw3 + 20), 178, cw3, v, l, sub, accent=acc)

tf = B.text(s, M, 300, W - M * 2, 120)
B.line(tf, "저희는 치료 결과를 약속하지 않습니다", 12, B.BLACK, B.XBOLD, first=True)
for t in [
    "「환자가 문의한 시점」과 「한국 의료진이 서류를 본 시점」 사이를 줄이는 일을 합니다.",
    "이 구간이 에이전시와 유치사업자가 실제로 통제할 수 있는 구간이고, 가장 자주 낭비되는 구간이기도 합니다.",
]:
    B.line(tf, t, 11, B.BODY, B.REG, before=8)

B.band(s, "▶ 저희가 줄이는 것은 치료 기간이 아니라 «치료를 시작하기까지의 시간»입니다")
B.note(s, "출처: 국가암정보센터 요약병기별 5년 상대생존율(2019~2023년 진단자). 연령표준화를 적용하지 않은 국내 기준 값이라 국제 비교표와 다를 수 있습니다. 인구 단위 통계이며 개인 예후는 담당 의료진이 판단합니다.")

# ── 그 시간이 어디서 새는가 — 위 칸은 «새는 자리», 아래 칸은 «저희가 바꾸는 자리»
#    (예전엔 아래를 현지 증언 3줄로 채웠는데, path 판의 같은 증언과 겹쳐
#     both 판에서 두 장에 그대로 반복됐다. 아래 40%가 비던 문제도 여기서 함께 닫는다.)
s = B.content(
    prs, "WHERE IT LEAKS",
    "그 시간은 대개 이렇게 샙니다",
    "위 칸은 환자가 실제로 지나는 경로이고, 아래 칸은 저희가 그 자리를 무엇으로 바꾸는지입니다.",
)
leaks = [
    (("① 지인·브로커에게 묻는다", ["메시지로 몇 주가 지나간다", "누가 책임지는지 불분명하다"]),
     ("등록 사업자가 창구를 맡습니다", ["보증보험을 갖춘 유치업자가", "접수부터 귀국까지 기록을 남깁니다"])),
    (("② 의무기록을 아무도 안 읽는다", ["번역도 정리도 안 된 채", "사진으로만 오간다"]),
     ("올리면 번역해 정리합니다", ["환자가 올린 기록을 정리해", "의료진이 볼 형태로 만듭니다"])),
    (("③ 약속으로 결정된다", ["의사 판단이 아니라", "「해줄 수 있다」는 말로"]),
     ("의료진 소견을 문서로 받습니다", ["진료과의 소견을 받아", "환자에게 문서로 전달합니다"])),
]
lw = (W - M * 2 - 33 * 2) / 3
for i, ((lt, ll), (rt, rl)) in enumerate(leaks):
    x = M + i * (lw + 33)
    B.step(s, x, 168, lw, 96, lt, ll, fill=B.PANEL)
    # 위 칸에서 아래 칸으로 내려가는 것을 세로 실선으로 잇는다
    B.box(s, x + lw / 2 - 0.7, 264, 1.4, 24, B.LINE)
    B.step(s, x, 288, lw, 100, rt, rl, fill=B.GREEN_L)

B.band(s, "▶ 저희가 파는 것은 병원 소개가 아니라 이 세 칸을 메우는 일입니다", y=404)
B.note(s, "위 칸은 저희가 접수한 문의와 현지 파트너 상담에서 확인한 경로이며, 표본을 갖춘 통계 조사 결과가 아닙니다. 아래 칸은 현재 플랫폼에서 실제로 운영하는 절차입니다.", y=452)

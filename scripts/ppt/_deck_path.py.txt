)
cw3 = (W - M * 2 - 20 * 2) / 3
for i, (v, l, sub, acc) in enumerate([
    ("3~4만 명", "연간 신규 암 진단", "출처에 따라 3만 또는 4만", False),
    ("약 6,000명", "치료 위해 해외로 출국", "자비 포함 · 목적지 전체", False),

]):
    B.stat(s, M + i * (cw3 + 20), 170, cw3, v, l, sub, accent=acc)
